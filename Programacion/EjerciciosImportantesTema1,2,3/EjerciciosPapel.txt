import java.util.Scanner;

public class Ejercicio1 {
	public static void main (String[]args) {
		Scanner sc = new Scanner (System.in);

		System.out.println("Introduce la temperatura en grados Celsius");
		
		double celsius = sc.nextDouble();

		if ( celsius < -100 || celsius > 3000) {
			System.out.println("Temperatura no aceptada");
		} else {
			double fahrenheit = (celsius * 9 / 5) + 32;
		
			double kelvin = celsius + 273.15;
			
			String mensajefahrenheit = "Fahrenheit =  " + fahrenheit;

			String mensajekelvin = "Kelvin =  " + kelvin;
			
			System.out.println(mensajefahrenheit);
			System.out.println(mensajekelvin);
		}
	}
}
